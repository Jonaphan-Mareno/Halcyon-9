    <script>
        // --- Game Setup & Audio ---
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        
        let cx = 0;
        let cy = 0;
        let maxRadius = 0;
        
        // Synths for sound effects
        const synths = {
            lock: new Tone.Synth({ oscillator: { type: "sine" }, envelope: { attack: 0.01, decay: 0.1, sustain: 0, release: 0.1 } }).toDestination(),
            miss: new Tone.MembraneSynth({ pitchDecay: 0.05, octaves: 2 }).toDestination(),
            win: new Tone.PolySynth(Tone.Synth, { oscillator: { type: "triangle" } }).toDestination(),
            lose: new Tone.FMSynth().toDestination()
        };

        // --- Game State ---
        const GameState = { MENU: 0, PLAYING: 1, GAMEOVER: 2 };
        let currentState = GameState.MENU;
        
        let rings = [];
        let activeRingIndex = 0;
        let lives = 3;
        let maxTime = 30;
        let timeLeft = maxTime;
        let lastTime = 0;
        let screenShake = 0;
        
        const targetAngle = Math.PI * 1.5; // 270 degrees (straight up)
        const targetTolerance = 0.18; // Radians of forgiveness (~10 degrees each side)
        const ringColors = ['#ff0055', '#ffaa00', '#00ffaa', '#00ccff', '#aa00ff'];

        // --- UI Elements ---
        const menus = {
            overlay: document.getElementById('menuOverlay'),
            main: document.getElementById('mainMenu'),
            gameOver: document.getElementById('gameOverMenu'),
            hud: document.getElementById('hud'),
            hint: document.getElementById('hint'),
            lives: document.getElementById('livesDisplay'),
            time: document.getElementById('timeDisplay'),
            endTitle: document.getElementById('endTitle'),
            endMsg: document.getElementById('endMessage')
        };

        // --- Initialization ---
        function resize() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
            cx = canvas.width / 2;
            cy = canvas.height / 2;
            // Determine max ring size based on screen
            maxRadius = Math.min(cx, cy) * 0.8;
            if (maxRadius > 300) maxRadius = 300; // Cap size on large screens
        }
        window.addEventListener('resize', resize);
        resize();

        function initLevel() {
            rings = [];
            const numRings = 4;
            const ringSpacing = maxRadius / (numRings + 1);
            
            for (let i = 0; i < numRings; i++) {
                // Rings start from outer to inner. Outer is index 0.
                let radius = maxRadius - (i * ringSpacing);
                
                // Speed gets generally faster or more tricky, alternating directions
                let baseSpeed = 0.02 + (Math.random() * 0.02) + (i * 0.015);
                let dir = (i % 2 === 0) ? 1 : -1;
                
                rings.push({
                    radius: radius,
                    angle: Math.random() * Math.PI * 2, // Random start
                    speed: baseSpeed * dir,
                    thickness: ringSpacing * 0.6,
                    color: ringColors[i % ringColors.length],
                    locked: false,
                    flash: 0 // For visual effect when locked
                });
            }
            
            activeRingIndex = 0;
            lives = 3;
            timeLeft = maxTime;
            updateHUD();
        }

        function startGame() {
            Tone.start(); // Required for browsers to allow audio
            initLevel();
            currentState = GameState.PLAYING;
            
            menus.overlay.classList.add('hidden');
            menus.main.classList.add('hidden');
            menus.gameOver.classList.add('hidden');
            menus.hud.classList.remove('hidden');
            menus.hint.classList.remove('hidden');
            
            lastTime = performance.now();
            requestAnimationFrame(gameLoop);
        }

        function gameOver(win) {
            currentState = GameState.GAMEOVER;
            menus.hud.classList.add('hidden');
            menus.hint.classList.add('hidden');
            menus.overlay.classList.remove('hidden');
            menus.gameOver.classList.remove('hidden');
            
            if (win) {
                menus.endTitle.textContent = "SYSTEM ALIGNED";
                menus.endTitle.className = "text-5xl font-black mb-4 uppercase tracking-widest text-green-400 neon-text";
                menus.endMsg.textContent = `Excellent timing. Time remaining: ${timeLeft.toFixed(1)}s`;
                synths.win.triggerAttackRelease(["C4", "E4", "G4", "C5"], "1n");
            } else {
                menus.endTitle.textContent = "ALIGNMENT FAILED";
                menus.endTitle.className = "text-5xl font-black mb-4 uppercase tracking-widest text-red-500 neon-text-red";
                menus.endMsg.textContent = "Core stability lost. Try again.";
                synths.lose.triggerAttackRelease("C2", "8n");
            }
        }

        // --- Input Handling ---
        function handleInput(e) {
            if (e.type === 'touchstart') e.preventDefault(); // Prevent double firing
            
            if (currentState !== GameState.PLAYING) return;
            
            const activeRing = rings[activeRingIndex];
            if (!activeRing || activeRing.locked) return;

            // Normalize angle to 0 - 2PI
            let normalizedAngle = activeRing.angle % (Math.PI * 2);
            if (normalizedAngle < 0) normalizedAngle += Math.PI * 2;
            
            // Calculate shortest angular distance to target
            let diff = Math.abs(normalizedAngle - targetAngle);
            if (diff > Math.PI) diff = Math.PI * 2 - diff;

            if (diff <= targetTolerance) {
                // Success - Lock it in dead center of target for visual crispness
                activeRing.angle = targetAngle; 
                activeRing.locked = true;
                activeRing.flash = 1.0;
                activeRingIndex++;
                
                synths.lock.triggerAttackRelease(activeRingIndex % 2 === 0 ? "C5" : "E5", "16n");

                // Check Win condition
                if (activeRingIndex >= rings.length) {
                    gameOver(true);
                }
            } else {
                // Miss
                lives--;
                screenShake = 15; // Initiate shake
                updateHUD();
                synths.miss.triggerAttackRelease("C2", "16n");

                if (lives <= 0) {
                    gameOver(false);
                }
            }
        }

        // Event listeners for clicking/tapping the game area
        window.addEventListener('pointerdown', handleInput);
        
        // Button handlers
        document.getElementById('startBtn').addEventListener('click', (e) => {
            e.stopPropagation();
            startGame();
        });
        document.getElementById('restartBtn').addEventListener('click', (e) => {
            e.stopPropagation();
            startGame();
        });

        function updateHUD() {
            // Update Lives
            menus.lives.innerHTML = '';
            for(let i=0; i<3; i++) {
                menus.lives.innerHTML += i < lives ? '●' : '○';
            }
            
            // Update Time
            menus.time.textContent = timeLeft.toFixed(1);
            if (timeLeft <= 5) {
                menus.time.classList.add('text-red-500', 'neon-text-red');
                menus.time.classList.remove('text-cyan-400', 'neon-text');
            } else {
                menus.time.classList.remove('text-red-500', 'neon-text-red');
                menus.time.classList.add('text-cyan-400', 'neon-text');
            }
        }

        // --- Rendering & Loop ---
        function drawTargetZone() {
            // Draw background indicator
            ctx.beginPath();
            ctx.moveTo(cx, cy);
            ctx.arc(cx, cy, maxRadius + 20, targetAngle - targetTolerance, targetAngle + targetTolerance);
            ctx.lineTo(cx, cy);
            ctx.fillStyle = 'rgba(0, 255, 255, 0.05)';
            ctx.fill();

            // Draw target bracket outer edge
            ctx.beginPath();
            ctx.arc(cx, cy, maxRadius + 15, targetAngle - targetTolerance, targetAngle + targetTolerance);
            ctx.lineWidth = 4;
            ctx.strokeStyle = 'rgba(0, 255, 255, 0.6)';
            ctx.stroke();
            
            // Draw line to center
            ctx.beginPath();
            ctx.moveTo(cx, cy - (maxRadius - (rings.length * (maxRadius/(rings.length+1)))) + 10);
            ctx.lineTo(cx, cy - maxRadius - 25);
            ctx.setLineDash([5, 5]);
            ctx.strokeStyle = 'rgba(0, 255, 255, 0.3)';
            ctx.lineWidth = 1;
            ctx.stroke();
            ctx.setLineDash([]);
        }

        function drawRings() {
            for (let i = 0; i < rings.length; i++) {
                const r = rings[i];
                const isActive = (i === activeRingIndex);
                
                // Track Background
                ctx.beginPath();
                ctx.arc(cx, cy, r.radius, 0, Math.PI * 2);
                ctx.lineWidth = r.thickness;
                ctx.strokeStyle = 'rgba(255, 255, 255, 0.03)';
                ctx.stroke();

                // If locked, draw full ring in color with lower opacity
                if (r.locked) {
                    ctx.beginPath();
                    ctx.arc(cx, cy, r.radius, 0, Math.PI * 2);
                    ctx.lineWidth = r.thickness;
                    ctx.strokeStyle = `${r.color}33`; // 20% opacity hex
                    ctx.stroke();
                }

                // Draw the node/indicator
                const nodeWidth = 0.2; // Radians
                ctx.beginPath();
                ctx.arc(cx, cy, r.radius, r.angle - nodeWidth/2, r.angle + nodeWidth/2);
                
                // Styling based on state
                if (r.locked) {
                    ctx.strokeStyle = '#ffffff';
                    ctx.shadowColor = '#ffffff';
                    ctx.shadowBlur = 10;
                } else if (isActive) {
                    ctx.strokeStyle = r.color;
                    ctx.shadowColor = r.color;
                    ctx.shadowBlur = 15;
                } else {
                    ctx.strokeStyle = 'rgba(255, 255, 255, 0.2)';
                    ctx.shadowBlur = 0;
                }
                
                ctx.lineWidth = r.thickness;
                ctx.stroke();
                ctx.shadowBlur = 0; // Reset shadow

                // Draw lock flash effect
                if (r.flash > 0) {
                    ctx.beginPath();
                    ctx.arc(cx, cy, r.radius, 0, Math.PI * 2);
                    ctx.lineWidth = r.thickness + (1 - r.flash) * 20;
                    ctx.strokeStyle = `rgba(255, 255, 255, ${r.flash * 0.5})`;
                    ctx.stroke();
                    r.flash -= 0.05;
                }
            }
        }

        function gameLoop(currentTime) {
            // Delta time calculation
            const dt = (currentTime - lastTime) / 1000;
            lastTime = currentTime;

            // Handle Screen Shake
            let drawCx = cx;
            let drawCy = cy;
            if (screenShake > 0) {
                drawCx += (Math.random() - 0.5) * screenShake;
                drawCy += (Math.random() - 0.5) * screenShake;
                screenShake *= 0.8; // Dampen
                if (screenShake < 0.5) screenShake = 0;
            }

            // Clear Screen
            ctx.fillStyle = '#050510';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Temporarily shift context for shake
            ctx.save();
            ctx.translate(drawCx - cx, drawCy - cy);

            if (currentState === GameState.PLAYING) {
                // Update Time
                timeLeft -= dt;
                if (timeLeft <= 0) {
                    timeLeft = 0;
                    gameOver(false);
                }
                updateHUD();

                // Update Ring Angles
                rings.forEach(r => {
                    if (!r.locked) {
                        // Frame-rate independent rotation (speed is approx base per frame * 60)
                        r.angle += r.speed * dt * 60; 
                    }
                });
            }

            drawTargetZone();
            drawRings();

            ctx.restore();

            // Continue loop
            if (currentState !== GameState.MENU) {
                requestAnimationFrame(gameLoop);
            }
        }
        
        // Initial drawing for menu background
        initLevel();
        drawTargetZone();
        drawRings();
        
    </script>
</body>
